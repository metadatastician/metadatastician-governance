# root-allow.txt — canonical allowlist for this repository's root.
#
# scripts/check-root-shape.sh fails when the root contains an entry that is
# not listed here. Derived from hyperpolymath/rsr-template-repo's allowlist,
# reduced to what a governance repository actually ships (no src/, no build
# system, no ABI seam) and extended with five justified additions, marked.
#
# Governance body documents live in docs/governance/ and architecture decision
# records in docs/decisions/ — both permitted by the RSR contract, which
# accepts governance at the root OR under docs/governance/.

README.adoc                # entry point; links Start here, Architecture decisions, Upstream pin
ARCHITECTURE.md            # repo layout + design rationale; RSR ships this at root
GOVERNANCE.md              # GitHub renders .md natively; RSR allows it at root
MAINTAINERS                # plain-text roster
SECURITY.md                # accepted at root OR .github/; GitHub's Security tab reads it here
LICENSE

.editorconfig

.gitignore

.machine_readable/         # descriptiles, binding templates, this allowlist
docs/                      # governance/ (body documents), decisions/ (ADRs)
scripts/                   # ci-health/ tooling, check-root-shape.sh

# --- Additions beyond the RSR list, specific to this repository -------------
AGENTS.md                  # agent orientation file. RSR allowlists the equivalent
                           # CLAUDE.md; this estate uses the AGENTS.md convention.
                           # Must stay at root: agents read it from the repo root.
templates/                 # project binding templates. This is the repository's
                           # product - individual projects bind through them - so
                           # it is not documentation and does not belong in docs/.
.oikos.yml                 # oikosbot configuration; the tool resolves it from the
                           # repository root, exactly like .hypatia-ignore in RSR.
TEST-NEEDS.adoc            # this repository's test-category declaration, required by
                           # the Testing & Benchmarking Taxonomy's Scope section
                           # ("a component must declare which categories apply and
                           # demonstrate evidence for each"). The estate convention puts
                           # it at the root (cf. proven-tests-and-benches/TEST-NEEDS.adoc),
                           # and it is a declaration of this repository, not governance
                           # body text, so it does not belong under docs/governance/.
tests/                     # shell test discovery is canonical at tests/*.sh
                           # (hyperpolymath/standards scripts/run-shell-test-suite.sh
                           # searches tests/ and scripts/tests/). Moving these under
                           # scripts/tests/ would work for the runner but would hide
                           # them from the convention every other repository follows.
